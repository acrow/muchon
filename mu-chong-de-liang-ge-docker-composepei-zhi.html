木憧的两个Docker-Compose配置

木憧官网

 

需要注意yml的版本，不同版本配置不同。和docker-compose的版本对应不是必须的，但可能会导致部分配置无效。

 

version: '2.4'
services:
  wordpress:
    depends_on:
    - db
    image: wordpress:latest
    volumes:
      - wordpress_files:/var/www/html
    ports:
    - "3080:80"
    restart: always
    cpus: '0.50'
    mem_limit: 100m
    environment:
      WORDPRESS_DB_HOST: db:3306
      WORDPRESS_DB_USER: wordpress
      WORDPRESS_DB_PASSWORD: wordpress
  db:
    image: mysql:5.7
    volumes:
      - db_data:/var/lib/mysql
    restart: always
    cpus: '0.50'
    mem_limit: 200m
    environment:
      MYSQL_ROOT_PASSWORD: whoami
      MYSQL_DATABASE: wordpress
      MYSQL_USER: wordpress
      MYSQL_PASSWORD: wordpress

volumes:
  wordpress_files:
  db_data:

 

来运动

 

version: '3.3'
services:
  laiyd:
    image: zhengchunchen/laiyd:v2
    ports:
    - "3001:3001"
    restart: always
    command: "/root/laiyd.sh"
    container_name: laiyd
