来运动Docker镜像

#!/bin/bash

er=$1
password=$2
if [ "$user"x = "x" ]
then
  user="acrow"
fi

if [ "$password"x = "x" ]
then
  password="whoami"
fi


#--------------------git------------------------#
git --version
if [ $? != 0 ]
then
  yum install -y git
fi

#--------------------nodejs------------------------#
node -v
if [ $? != 0 ]
then
  curl -sL https://rpm.nodesource.com/setup_10.x | bash -
  yum install -y nodejs
fi
#--------------------pm2------------------------#
pm2 -version
if [ $? != 0 ]
then
  npm config set registry https://registry.npm.taobao.org
  npm install -g pm2
fi

#--------------------mongodb------------------------#
mongo --version
if [ $? != 0 ]
then
  echo "[mongodb-org-3.6]" > /etc/yum.repos.d/mongodb-org-3.6.repo
  echo "name=MongoDB Repository" >> /etc/yum.repos.d/mongodb-org-3.6.repo
  echo "baseurl=https://repo.mongodb.org/yum/redhat/7/mongodb-org/3.6/x86_64/" >> /etc/yum.repos.d/mongodb-org-3.6.repo
  echo "gpgcheck=1" >> /etc/yum.repos.d/mongodb-org-3.6.repo
  echo "enabled=1" >> /etc/yum.repos.d/mongodb-org-3.6.repo
  echo "gpgkey=https://www.mongodb.org/static/pgp/server-3.6.asc" >> /etc/yum.repos.d/mongodb-org-3.6.repo
  yum install -y mongodb-org
fi

#--------------------laiyd-www------------------------#
cd /root/www
if [ $? != 0 ]
then
  cd /root
  mkdir www
  cd www
fi
cd laiyd
if [ $? != 0 ]
then
  git clone https://$user:$password@git.coding.net/acrow/laiyd.git
else
  git pull
fi
pm2 stop laiyd
cd /root/www/laiyd/bin
pm2 start www --name laiyd
pm2 stop laiydbatch
cd /root/www/laiyd/bin2
pm2 start app.js --name laiydbatch

#--------------------laiyd-db------------------------#
cd /root/mongo
if [ $? != 0 ]
then
  cd /root
  mkdir mongo
  cd mongo
fi
cd data
if [ $? != 0 ]
then
  mkdir data
fi
mongod --dbpath=/root/mongo/data --logpath=/root/mongo/laiyd.log --logappend
